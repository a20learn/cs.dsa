Chapter 02 // Linear Data Structures

Stacks & Queues

1. The Stack (LIFO Mechanics)

A Stack is a sequential data structure that operates strictly on a Last-In, First-Out basis. Elements can only be added or removed from the top layer, mirroring how the CPU manages function calls natively inside hardware memory frames.

Operational Costs

  • Push (Insertion at Top): O(1) complexity.
  • Pop (Removal from Top): O(1) complexity.
  • Peek (Look at Top): O(1) complexity.

Stack Blueprint

type Stack struct {
    items []int
}

func (s *Stack) Push(val int) { s.items = append(s.items, val) }
func (s *Stack) Pop() int {
    if len(s.items) == 0 { return -1 }
    res := s.items[len(s.items)-1]
    s.items = s.items[:len(s.items)-1]
    return res
}
#define MAX 100
struct Stack {
    int top;
    int items[MAX];
};

void push(struct Stack* s, int val) {
    if (s->top < MAX - 1) s->items[++(s->top)] = val;
}
int pop(struct Stack* s) {
    return (s->top == -1) ? -1 : s->items[(s->top)--];
}
import java.util.ArrayList;

class Stack {
    private ArrayList<Integer> items = new ArrayList<>();

    public void push(int val) { items.add(val); }
    public int pop() {
        if (items.isEmpty()) return -1;
        return items.remove(items.size() - 1);
    }
}
class Stack:
    def __init__(self):
        self.items = []

    def push(self, val):
        self.items.append(val)

    def pop(self):
        return self.items.pop() if self.items else -1

2. The Queue (FIFO Mechanics)

A Queue is a linear data structure that processes elements sequentially on a First-In, First-Out basis. Data enters through the back boundary (Enqueue) and exits exclusively from the front boundary (Dequeue).

Operational Costs

  • Enqueue (Insert at Tail): O(1) complexity.
  • Dequeue (Remove from Head): O(1) complexity.
  • Front (Look at Head): O(1) complexity.

Queue Blueprint

type Queue struct {
    items []int
}

func (q *Queue) Enqueue(val int) { q.items = append(q.items, val) }
func (q *Queue) Dequeue() int {
    if len(q.items) == 0 { return -1 }
    res := q.items[0]
    q.items = q.items[1:]
    return res
}
#define MAX 100
struct Queue {
    int front, rear;
    int items[MAX];
};

void enqueue(struct Queue* q, int val) {
    if (q->rear < MAX - 1) q->items[++(q->rear)] = val;
}
int dequeue(struct Queue* q) {
    if (q->front > q->rear) return -1;
    return q->items[(q->front)++];
}
import java.util.LinkedList;

class Queue {
    private LinkedList<Integer> items = new LinkedList<>();

    public void enqueue(int val) { items.addLast(val); }
    public int dequeue() {
        return items.isEmpty() ? -1 : items.removeFirst();
    }
}
from collections import deque

class Queue:
    def __init__(self):
        self.items = deque()

    def enqueue(self, val):
        self.items.append(val)

    def dequeue(self):
        return self.items.popleft() if self.items else -1
